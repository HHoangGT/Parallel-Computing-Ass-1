if(TSAN)
  set(test_locks_args --iters 2000 --maxt 8)
  set(test_barrier_args --episodes 2000 --maxt 8)
  set(test_counter_args --per 10000 --maxt 8)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(test_runner setarch ${CMAKE_SYSTEM_PROCESSOR} -R)
  endif()
endif()

file(GLOB test_sources CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
foreach(source ${test_sources})
  get_filename_component(name ${source} NAME_WE)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE common)
  add_test(NAME ${name} COMMAND ${test_runner} $<TARGET_FILE:${name}> ${${name}_args})
endforeach()
